# ATD in voice mode (V.253 9.1.1), on a line that does not tell when the
# call is answered: OK once +VRN has gone without a ringing tone
A AT+FCLASS=8\r
A AT+VRA?\r
A AT+VRN?\r
A AT+VRN=?\r
A AT+VRN=256\r
A AT+VRN=1\r
A ATD5551000\r
run 1500
A AT+VLS?\r
# the line taken: a number dialled on it is DTMF digits, OK after them
A ATS11=50\r
A ATD12\r
run 400
# ATH in voice mode: on hook, and data mode again (V.253 9.1.2)
A ATH\r
A AT+FCLASS?\r
# +VRN beyond S7: S7 only, NO ANSWER when it has gone
A AT+FCLASS=8\r
A ATS7=2\r
A AT+VRN=5\r
A ATD5551000\r
run 2500
A AT+VLS?\r
# ATDL: the last number again
A AT+VRN=0\r
A ATDL\r
run 200
A AT+VLS=0\r
# a ringing tone (North America's, 2 s on and 4 s off): <DLE>r as each
# ring is heard, and +VRN no longer applies; OK when +VRA has gone since
# the last ring ended (V.253 10.2.5)
A AT+VRN=1\r
A AT+VRA=50\r
A ATS7=60\r
tone A 440+480 2000 4000
A ATD5551000\r
run 8000
tone A off
run 6000
A AT+VLS=0\r
# +VRA shorter than the gap between rings: OK in it
A AT+VRA=30\r
tone A 440+480 2000 4000
A ATD5551000\r
run 6000
tone A off
A AT+VLS=0\r
# +VRA=0: OK as soon as the first ring has ended
A AT+VRA=0\r
tone A 425 1000 4000
A ATD5551000\r
run 2000
tone A off
A AT+VLS=0\r
# ringing that goes on: NO ANSWER after S7
A AT+VRA=50\r
A ATS7=8\r
tone A 425 1000 4000
A ATD5551000\r
run 9000
tone A off
# a busy tone's cadence: <DLE>b and BUSY
A AT+VRN=10\r
tone A 480+620 500 500
A ATD5551000\r
run 5000
tone A off
A AT+VLS?\r
